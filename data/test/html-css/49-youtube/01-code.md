title: |-
  ①
  「 <!-- ここに埋め込み --> 」のところに
  YouTubeの動画を、なんでもいいので探してきて
  1つ埋め込みましょう。
language: html
required: true
starter: |-
  <!DOCTYPE html>
  <html>
      <head></head>
      <body>
          <div class="wrapper">
              <!-- ここに埋め込み -->
          </div>
          <style>
              .wrapper {
                  width: 80vw;
                  margin: 0 auto;
              }
          </style>
      </body>
  </html>
example: |-
  <!DOCTYPE html>
  <html>
      <head></head>
      <body>
          <div class="wrapper">
              <iframe src="https://www.youtube.com/embed/aqz-KE-bpKQ" title="動画"></iframe>
          </div>
          <style>
              .wrapper {
                  width: 80vw;
                  margin: 0 auto;
              }
          </style>
      </body>
  </html>
